冻结 Patch 编码器，检查图文条件分类的梯度路径

18_multimodal_demo.py
# [1] 随机初始化的接口教学，不代表已训练好的 VLMimport torchfrom torch import nnfrom torch.nn import functional as Ftorch.manual_seed(16)B, D = 2, 24images = torch.randn(B, 3, 16, 16)vision = nn.Conv2d(3, 12, kernel_size=4, stride=4)vision.requires_grad_(False)vision.eval()projector = nn.Linear(12, D)patch_position = nn.Parameter(torch.randn(1, 16, D) * 0.02)text_embedding = nn.Embedding(32, D)fusion = nn.MultiheadAttention(D, 4, dropout=0.0, batch_first=True)classifier = nn.Linear(D, 2) # [2] 输入前没有可训练模块；只关闭固定视觉编码器的图记录def classify(images, token_ids):    with torch.no_grad():        raw = vision(images).flatten(2).transpose(1, 2)    visual = projector(raw) + patch_position    # 固定长度、无 padding 的教学文本；真实变长输入需要有效位    query = text_embedding(token_ids).mean(dim=1, keepdim=True)    pooled, _ = fusion(query, visual, visual, need_weights=False)    return classifier(pooled[:, 0]), visual, query # [3] 两类合成标签使用整数类别，CrossEntropy 接收原始 logitstoken_ids = torch.tensor([[2, 4, 7], [2, 9, 7]], dtype=torch.long)labels = torch.tensor([0, 1], dtype=torch.long)logits, visual, query = classify(images, token_ids)loss = F.cross_entropy(logits, labels)loss.backward()assert logits.shape == (B, 2)assert all(p.grad is None for p in vision.parameters())tracked = [projector.weight, patch_position, text_embedding.weight,           fusion.in_proj_weight, classifier.weight]assert all(p.grad is not None and torch.isfinite(p.grad).all() for p in tracked)assert all(p.grad.norm() > 0 for p in tracked) # [4] 固定图像，只替换一个样本的文本，检查条件路径与样本隔离changed_ids = token_ids.clone()changed_ids[0, 1] = 11with torch.no_grad():    changed, _, _ = classify(images, changed_ids)    torch.testing.assert_close(changed[1], logits.detach()[1])    assert not torch.allclose(changed[0], logits.detach()[0])print("视觉 / 查询 / 分类 logits:", visual.shape, query.shape, logits.shape)print("冻结边界、可训练梯度和文本条件路径检查通过")

从特殊 token 中正确抽取空间特征

用已编码的合成张量检查 CLS、Register 与行优先 Patch 的切片、非方形网格重排及梯度。无权重下载；该实验验证读取约定，不模拟 DINOv2 的训练或评价其表示质量。

运行时应核对

两种布局分别输出 tokens=(2,7,8) 与 (2,11,8)，都恢复为 grid=(2,8,2,3)。空间顺序、读取层梯度及错误元数据的断言通过；LayerNorm 输出和单位 L2 范数不混同。

multimodal_visual-token-contract.py
import torchimport torch.nn.functional as F torch.manual_seed(52) def split_visual_tokens(hidden, grid_hw, num_registers):    """Contract: [CLS, registers, row-major patches], all already encoded."""    if hidden.ndim != 3:        raise ValueError("expected [B,N,D]")    gh, gw = grid_hw    if gh <= 0 or gw <= 0 or num_registers < 0:        raise ValueError("invalid grid or register count")    start = 1 + num_registers    if hidden.shape[1] != start + gh * gw:        raise ValueError("token count disagrees with the declared layout")    patches = hidden[:, start:]    grid = patches.transpose(1, 2).reshape(hidden.shape[0], hidden.shape[2], gh, gw)    return {"cls": hidden[:, 0], "registers": hidden[:, 1:start],            "patches": patches, "grid": grid} for registers in (0, 4):    # A non-square grid catches implementations that infer sqrt(N) blindly.    x = torch.randn(2, 1 + registers + 6, 8, dtype=torch.float64,                    requires_grad=True)    features = split_visual_tokens(x, (2, 3), registers)    assert features["cls"].shape == (2, 8)    assert features["registers"].shape == (2, registers, 8)    assert features["grid"].shape == (2, 8, 2, 3)    torch.testing.assert_close(features["grid"].flatten(2).transpose(1, 2),                               features["patches"])    torch.testing.assert_close(features["grid"][1, :, 1, 2],                               x[1, 1 + registers + 5])    features["grid"].mean(dim=(2, 3)).sum().backward()    expected = torch.zeros_like(x)    expected[:, 1 + registers:] = 1 / 6    torch.testing.assert_close(x.grad, expected)    print(f"registers={registers}: tokens={tuple(x.shape)}, grid={tuple(features['grid'].shape)}") for grid_hw, registers in [((2, 3), 0), ((2, 2), 4), ((0, 3), 4)]:    try:        split_visual_tokens(x, grid_hw, registers)        raise AssertionError("incorrect metadata must be rejected")    except ValueError:        pass # LayerNorm over channels is a different operation from unit-length scaling.ln = F.layer_norm(x.detach(), (8,))unit = F.normalize(ln, p=2, dim=-1)torch.testing.assert_close(ln.mean(dim=-1), torch.zeros_like(ln[..., 0]), atol=1e-12, rtol=0)torch.testing.assert_close(unit.norm(dim=-1), torch.ones_like(unit[..., 0]))assert not torch.allclose(ln.norm(dim=-1), unit.norm(dim=-1))print("passed: spatial order, prefix gradient exclusion, layout errors, LayerNorm vs L2")
