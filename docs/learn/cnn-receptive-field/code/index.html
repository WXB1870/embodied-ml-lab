完整小 CNN 训练与 Patch 投影等价验证

保留原有条纹分类网络与训练设置，先逐层追踪形状、1126 个参数、每样本 155544 MAC 及激活/参数梯度；验证局部感受野和 GAP 梯度，再完成训练、留出评价及 patch 投影等价对照。

运行时应核对

固定条纹任务的训练交叉熵降低，参数发生更新，1126 个参数；卷积与 Linear patch token 数值相同。留出表现仅描述这个合成分布。 额外检查每层激活/参数梯度有限、GAP 反向均分以及局部卷积输出窗外梯度为零；卷积和 Linear 的单样本 MAC 共 155544，不包含激活、池化、偏置与反向开销。

import copyimport torchfrom torch import nnfrom torch.nn import functional as F torch.manual_seed(31)torch.set_num_threads(1) # 不依赖外部图像：RGB 三通道共享同一条亮线，颜色不作为标签捷径。count, side = 160, 12images = .04 * torch.randn(count, 3, side, side)labels = torch.arange(count) % 2  # 0=竖线，1=横线。positions = torch.randint(3, 8, (count,))for index, position in enumerate(positions.tolist()):    if labels[index] == 0:        images[index, :, 2:10, position:position + 2] += 1.    else:        images[index, :, position:position + 2, 2:10] += 1.order = torch.randperm(count)images, labels = images[order], labels[order]train_x, test_x = images[:120], images[120:]train_y, test_y = labels[:120], labels[120:]assert train_x.shape == (120, 3, 12, 12) class SmallCNN(nn.Module):    def __init__(self):        super().__init__()        self.features = nn.Sequential(            nn.Conv2d(3, 8, 3, padding=1), nn.ReLU(),            nn.Conv2d(8, 12, 3, padding=1), nn.ReLU())        self.pool = nn.AdaptiveAvgPool2d(1)        self.classifier = nn.Linear(12, 2)     def forward(self, x):        feature_map = self.features(x)  # B,12,H,W；两层卷积的局部感受野为 5×5。        pooled = self.pool(feature_map).flatten(1)        return self.classifier(pooled)  # 原始 logits，不提前 Softmax。 model = SmallCNN()# 在同一个 SmallCNN 上追踪前向、参数量、MACs 与反向，不另建架构。def trace_network(original, sample, target):    network = copy.deepcopy(original)    x = sample.detach().clone().requires_grad_()    values = [("input", x)]    h = x    total_parameters, total_macs = 0, 0    r, j = 1, 1    for index, layer in enumerate(network.features):        h = layer(h)        h.retain_grad()        values.append((f"features.{index}", h))        parameters = sum(p.numel() for p in layer.parameters())        macs = 0        if isinstance(layer, nn.Conv2d):            k, s, d = layer.kernel_size[0], layer.stride[0], layer.dilation[0]            r += (k - 1) * d * j            j *= s            macs = h.shape[-2] * h.shape[-1] * layer.out_channels * layer.in_channels * k * k        total_parameters += parameters        total_macs += macs        print(values[-1][0], tuple(h.shape), "params", parameters, "MAC/sample", macs, "r,j", r, j)    assert (r, j) == (5, 1)    pooled = network.pool(h).flatten(1)    pooled.retain_grad()    logits = network.classifier(pooled)    logits.retain_grad()    loss = F.cross_entropy(logits, target)    loss.backward()    # CE -> Linear -> GAP，梯度在每个通道的 144 个位置均分。    upstream = logits.detach().softmax(-1)    upstream[torch.arange(len(target)), target] -= 1    upstream /= len(target)    torch.testing.assert_close(logits.grad, upstream)    expected_pooled = upstream @ network.classifier.weight.detach()    torch.testing.assert_close(pooled.grad, expected_pooled)    torch.testing.assert_close(h.grad, (expected_pooled[:, :, None, None] / 144).expand_as(h))    for name, value in values:        assert value.grad is not None and torch.isfinite(value.grad).all()        print("activation gradient", name, "shape", tuple(value.grad.shape), "norm", value.grad.norm().item())    for name, parameter in network.named_parameters():        assert parameter.grad is not None and torch.isfinite(parameter.grad).all()        print("parameter gradient", name, tuple(parameter.grad.shape), parameter.grad.norm().item())    total_parameters += sum(p.numel() for p in network.classifier.parameters())    total_macs += 12 * 2    assert total_parameters == 1126 and total_macs == 155544    # 此处考察局部卷积输出，不是含 GAP 的分类 loss。    local_x = sample[:1].detach().clone().requires_grad_()    local = network.features[:3](local_x)    local_gradient, = torch.autograd.grad(local[0, :, 6, 6].sum(), local_x)    outside = torch.ones_like(local_gradient, dtype=torch.bool)    outside[:, :, 4:9, 4:9] = False    assert torch.count_nonzero(local_gradient[outside]) == 0    assert torch.count_nonzero(local_gradient) > 0    print("local output support within 5x5; total params=1126, MAC/sample=155544") trace_network(model, train_x[:4], train_y[:4])optimizer = torch.optim.Adam(model.parameters(), lr=.01)with torch.no_grad():    initial = F.cross_entropy(model(train_x), train_y).item()original_weight = model.features[0].weight.detach().clone()model.train()for step in range(100):    optimizer.zero_grad(set_to_none=True)    logits = model(train_x)    loss = F.cross_entropy(logits, train_y)    loss.backward()    assert all(p.grad is not None and torch.isfinite(p.grad).all() for p in model.parameters())    optimizer.step() model.eval()with torch.no_grad():    final = F.cross_entropy(model(train_x), train_y).item()    holdout_logits = model(test_x)    holdout_loss = F.cross_entropy(holdout_logits, test_y).item()    holdout_accuracy = (holdout_logits.argmax(-1) == test_y).float().mean().item()    assert model(torch.randn(2, 3, 10, 14)).shape == (2, 2)assert final < initial * .3  # 仅验证固定条纹任务的训练链路。assert not torch.equal(original_weight, model.features[0].weight)assert sum(p.numel() for p in model.parameters()) == 1126print(f"train CE: {initial:.6f} -> {final:.6f}")print(f"holdout CE: {holdout_loss:.6f}, accuracy: {holdout_accuracy:.3f}") # 再验证 ViT 常见的 patch embedding：不重叠 patch 的共享线性映射。patch_size, embedding_dim = 4, 7patch_layer = nn.Conv2d(3, embedding_dim, patch_size, stride=patch_size)sample = test_x[:2]conv_tokens = patch_layer(sample).flatten(2).transpose(1, 2)patches = F.unfold(sample, kernel_size=patch_size, stride=patch_size).transpose(1, 2)linear_tokens = F.linear(patches, patch_layer.weight.flatten(1), patch_layer.bias)assert conv_tokens.shape == (2, 9, 7)torch.testing.assert_close(conv_tokens, linear_tokens, rtol=1e-5, atol=1e-6)# 第一片按 C、局部 H、局部 W 展平，与 F.unfold 的列顺序一致。torch.testing.assert_close(patches[0, 0], sample[0, :, :4, :4].reshape(-1))print("通过：卷积分类训练、独立留出评价、参数量与 patch 线性投影等价。")print("此处输入尺寸可变仅是形状性质，不代表对新分辨率的泛化保证。")

用 Unfold 手写分组卷积并比较三组梯度

一次覆盖普通卷积、分组卷积和 depthwise multiplier=2；非方形输入/核与不相等的空间参数用于暴露实现错误。

运行时应核对

三种分组的输出均为 [2,Cout,3,8]，前向与三组梯度一致；32→64 的 3×3 卷积有 18496 个参数。

cnn_unfold-convolution.py
import torchfrom torch.nn import functional as F torch.manual_seed(23)torch.set_num_threads(1) # 教学实现：支持二维整数 tuple、对称零填充和分组，不处理字符串 padding。def pair(value):    return (value, value) if isinstance(value, int) else tuple(value) def unfold_conv(x, weight, bias=None, stride=1, padding=0, dilation=1, groups=1):    sh, sw = pair(stride)    ph, pw = pair(padding)    dh, dw = pair(dilation)    batch, channels, height, width = x.shape    out_channels, per_group, kh, kw = weight.shape    if groups <= 0 or channels % groups or out_channels % groups:        raise ValueError("输入输出通道必须能被 groups 整除")    if per_group != channels // groups:        raise ValueError("权重第二维必须等于 Cin/groups")    if min(sh, sw, dh, dw, kh, kw) <= 0 or min(ph, pw) < 0:        raise ValueError("卷积几何参数无效")    oh = (height + 2 * ph - dh * (kh - 1) - 1) // sh + 1    ow = (width + 2 * pw - dw * (kw - 1) - 1) // sw + 1    if min(oh, ow) <= 0:        raise ValueError("没有有效输出位置")    # B,Cin*Kh*Kw,L：最后一轴是输出位置，每列是一个局部窗口。    columns = F.unfold(x, (kh, kw), dilation=(dh, dw), padding=(ph, pw), stride=(sh, sw))    assert columns.shape[-1] == oh * ow    grouped_columns = columns.reshape(batch, groups, per_group * kh * kw, oh * ow)    grouped_weights = weight.reshape(groups, out_channels // groups, per_group * kh * kw)    output = torch.einsum("goc,bgcl->bgol", grouped_weights, grouped_columns)    output = output.reshape(batch, out_channels, oh, ow)    if bias is not None:        if bias.shape != (out_channels,):            raise ValueError("每个输出通道需要一个偏置")        output = output + bias[None, :, None, None]    return output # 普通卷积、分组卷积、depthwise multiplier=2；均采用非方形核。cases = [(1, 3, 5), (2, 4, 6), (4, 4, 8)]for groups, cin, cout in cases:    x = torch.randn(2, cin, 7, 9, dtype=torch.float64, requires_grad=True)    weight = torch.randn(cout, cin // groups, 3, 2, dtype=torch.float64, requires_grad=True)    bias = torch.randn(cout, dtype=torch.float64, requires_grad=True)    options = dict(stride=(2, 1), padding=(1, 0), dilation=(2, 1), groups=groups)    actual = unfold_conv(x, weight, bias, **options)    reference = F.conv2d(x, weight, bias, **options)    assert actual.shape == (2, cout, 3, 8)    torch.testing.assert_close(actual, reference, rtol=1e-10, atol=1e-10)    # 相同上游权重对应一个明确的向量—Jacobian 乘积。    upstream = torch.linspace(.1, 1., actual.numel(), dtype=x.dtype).reshape_as(actual)    actual_grads = torch.autograd.grad((actual * upstream).sum(), (x, weight, bias))    reference_grads = torch.autograd.grad((reference * upstream).sum(), (x, weight, bias))    for ours, expected in zip(actual_grads, reference_grads):        torch.testing.assert_close(ours, expected, rtol=1e-10, atol=1e-10)    print("groups =", groups, "shape =", tuple(actual.shape), "forward and gradients agree") # 空间共享参数：固定一个卷积核，图像尺寸不进入参数总数。layer = torch.nn.Conv2d(32, 64, kernel_size=3, bias=True)assert sum(p.numel() for p in layer.parameters()) == 18496# Unfold 会显式展开重叠窗口；这是理解用实现，不是性能优化建议。example = torch.randn(1, 3, 16, 16)expanded = F.unfold(example, kernel_size=3, padding=1)print("unfold/input element ratio =", expanded.numel() / example.numel())assert expanded.numel() == example.numel() * 9print("通过：互相关、通道分组、非方形几何、输入/权重/偏置梯度。")
